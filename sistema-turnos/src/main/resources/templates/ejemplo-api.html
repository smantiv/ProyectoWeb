📝 Código de Ejemplo (Copiar y Usar)

// En tu archivo HTML o JavaScript:

// 1. Cargar los scripts
<script src="/js/api/ApiClient.js"></script>
<script src="/js/api/UsuarioService.js"></script>
<script src="/js/api/TurnoService.js"></script>
<script src="/js/api/ApiServices.js"></script>

// 2. Usar los servicios
async function ejemplo() {
    try {
        // Obtener todos los turnos
        const turnos = await ApiServices.turnoService.obtenerTodos();
        console.log('Turnos:', turnos);

        // Crear nuevo turno
        const nuevoTurno = await ApiServices.turnoService.crear({
            fecha: '2024-03-24',
            horaInicio: '08:00:00',
            horaFin: '12:00:00',
            estado: 'disponible',
            zonaId: 1
        });
        console.log('Turno creado:', nuevoTurno);

        // Obtener docentes
        const docentes = await ApiServices.docenteService.obtenerTodos();
        console.log('Docentes:', docentes);

        // Actualizar usuario
        const usuarioActualizado = await ApiServices.usuarioService.actualizar(1, {
            nombre: 'Nuevo nombre',
            activo: true
        });
        console.log('Usuario actualizado:', usuarioActualizado);

    } catch (error) {
        console.error('Error:', error.message);
        console.error('Detalles:', error.responseData);
    }
}

// 3. Ejecutar
ejemplo();
